package com.kuli.music;

import android.app.Activity;
import android.os.Bundle;
import android.graphics.Color;
import android.view.Gravity;
import android.widget.LinearLayout;
import android.widget.TextView;

public class MainActivity extends Activity {
    @Override
    public void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        LinearLayout pantalla = new LinearLayout(this);
        pantalla.setOrientation(LinearLayout.VERTICAL);
        pantalla.setGravity(Gravity.CENTER);
        pantalla.setBackgroundColor(Color.rgb(18,18,18));

        TextView titulo = new TextView(this);
        titulo.setText("🎵 MUSIXA 🎵");
        titulo.setTextSize(30);
        titulo.setTextColor(Color.WHITE);
        titulo.setGravity(Gravity.CENTER);

        pantalla.addView(titulo);
        android.widget.Button buscar = new android.widget.Button(this);
        buscar.setText("🎵 BUSCAR MÚSICA");
        buscar.setOnClickListener(v -> {
            android.content.Intent i = new android.content.Intent(
                android.content.Intent.ACTION_VIEW,
                android.net.Uri.parse("https://www.jamendo.com/search"));
            startActivity(i);
        });
        pantalla.addView(buscar);
        
        android.widget.LinearLayout controles = new android.widget.LinearLayout(this);
        controles.setGravity(android.view.Gravity.CENTER);

        android.widget.Button play = new android.widget.Button(this);
        play.setText("▶️");
        play.setOnClickListener(v -> {
            if (reproductor != null && !reproductor.isPlaying())
                reproductor.start();
        });

        android.widget.Button pausa = new android.widget.Button(this);
        pausa.setText("⏸️");
        pausa.setOnClickListener(v -> {
            if (reproductor != null && reproductor.isPlaying())
                reproductor.pause();
        });

        android.widget.Button parar = new android.widget.Button(this);
        parar.setText("⏹️");
        parar.setOnClickListener(v -> {
            if (reproductor != null) {
                reproductor.pause();
                reproductor.seekTo(0);
            }
        });

        controles.addView(play);
        controles.addView(pausa);
        controles.addView(parar);
        pantalla.addView(controles);

        
        android.widget.Button lista = new android.widget.Button(this);
        lista.setText("🎶 MIS CANCIONES");
        lista.setOnClickListener(v -> {
            android.content.Intent i = new android.content.Intent(
                android.content.Intent.ACTION_OPEN_DOCUMENT);
            i.setType("audio/*");
            i.addCategory(android.content.Intent.CATEGORY_OPENABLE);
            i.putExtra(android.content.Intent.EXTRA_ALLOW_MULTIPLE, true);
            startActivityForResult(i, 1);
        });
        pantalla.addView(lista);

        
        android.widget.Button verCanciones = new android.widget.Button(this);
        verCanciones.setBackgroundTintList(android.content.res.ColorStateList.valueOf(android.graphics.Color.rgb(30,215,96)));
        verCanciones.setText("📂 VER CANCIONES DEL MÓVIL");
        verCanciones.setOnClickListener(v -> {
            android.content.Intent i = new android.content.Intent(
                android.content.Intent.ACTION_OPEN_DOCUMENT);
            i.setType("audio/*");
            i.addCategory(android.content.Intent.CATEGORY_OPENABLE);
            startActivityForResult(i, 1);
        });
        pantalla.addView(verCanciones);

        for (int n = 0; n < pantalla.getChildCount(); n++) { android.view.View v = pantalla.getChildAt(n); if (v instanceof android.widget.Button) { v.setBackgroundTintList(android.content.res.ColorStateList.valueOf(android.graphics.Color.rgb(30,215,96))); } }
        setContentView(pantalla);
    }

    private android.media.MediaPlayer reproductor;

    @Override
    protected void onActivityResult(int requestCode, int resultCode, android.content.Intent data) {
        super.onActivityResult(requestCode, resultCode, data);
        if (requestCode == 1 && resultCode == RESULT_OK && data != null) {
            try {
                if (reproductor != null) reproductor.release();
                reproductor = android.media.MediaPlayer.create(this, data.getData());
                if (reproductor == null) throw new Exception("No se pudo abrir");
                reproductor.start();
                android.widget.Toast.makeText(this, "Reproduciendo música", 0).show();
            } catch (Exception e) {
                android.widget.Toast.makeText(this, "Error: " + e.getMessage(), 1).show();
            }
        }
    }

}
